#!/usr/bin/env python3

import argparse
import json
import socket


def check_port(host, port, timeout):
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
        sock.settimeout(timeout)
        return sock.connect_ex((host, port)) == 0


def main():
    parser = argparse.ArgumentParser(
        description="ReconX - reconocimiento TCP autorizado"
    )

    parser.add_argument("target")
    parser.add_argument("--ports", default="22,53,80,443,8080")
    parser.add_argument("--timeout", type=float, default=1.0)
    parser.add_argument("-o", "--output", default="reconx_report.json")

    args = parser.parse_args()

    try:
        ip = socket.gethostbyname(args.target)
    except socket.gaierror:
        raise SystemExit("[!] No se pudo resolver el objetivo")

    ports = []

    try:
        for value in args.ports.split(","):
            port = int(value.strip())

            if not 1 <= port <= 65535:
                raise ValueError

            ports.append(port)

    except ValueError:
        raise SystemExit("[!] Lista de puertos inválida")

    open_ports = []

    for port in ports:
        if check_port(ip, port, args.timeout):
            open_ports.append(port)
            print(f"[OPEN] {port}")

    report = {
        "tool": "ReconX",
        "target": args.target,
        "ip": ip,
        "open_ports": open_ports,
    }

    with open(args.output, "w", encoding="utf-8") as f:
        json.dump(report, f, indent=2, ensure_ascii=False)

    print(f"[+] Puertos abiertos: {len(open_ports)}")
    print(f"[+] Reporte: {args.output}")
